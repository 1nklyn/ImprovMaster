// Formats seconds into MM:SS
const formatTime = (seconds) => {
  const m = Math.floor(seconds / 60);
  const s = seconds % 60;
  return `${m}:${s.toString().padStart(2, '0')}`;
};

// Calls Gemini API to grade the speech
const evaluateSpeech = async (topic, transcript, type, goals) => {
  const apiKey = ""; // Canvas provides this automatically
  const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-3-flash-preview:generateContent?key=${apiKey}`;

  const goalsText = goals && goals.length > 0 
    ? `\n\nAdditionally, specifically evaluate the speaker against their personal goals: ${goals.map(g => g.text).join(', ')}. Mention how well they adhered to these goals in your summary.` 
    : '';

  const prompt = type === 'quickYap' 
    ? `Evaluate this 1-minute improv speech about: "${topic}".${goalsText}`
    : `Evaluate this roleplay response to the scenario: "${topic}".${goalsText}`;

  const payload = {
    contents: [{ parts: [{ text: `${prompt}\n\nTranscript:\n"${transcript || '(No transcript recorded)'}"` }] }],
    generationConfig: {
      responseMimeType: "application/json",
      responseSchema: {
        type: "OBJECT",
        properties: {
          grade: { type: "STRING", description: "A letter grade (A+, A, B, C, D, or F) based on coherence, relevance, and fluency. If empty transcript, give F." },
          summary: { type: "STRING", description: "A 2-3 sentence constructive feedback summary." }
        },
        required: ["grade", "summary"]
      }
    },
    systemInstruction: {
      parts: [{ text: "You are an expert speech and improv coach. You give fair, constructive feedback. You evaluate based on relevance to the topic, flow of ideas, and overall coherence." }]
    }
  };

  try {
    const response = await fetch(apiUrl, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(payload)
    });
    const data = await response.json();
    const text = data.candidates?.[0]?.content?.parts?.[0]?.text;
    if (text) {
      return JSON.parse(text);
    }
  } catch (error) {
    console.error("Gemini API Error:", error);
  }
  return { grade: "N/A", summary: "Failed to generate feedback. Please try again later." };
};

export default function App() {
  const [user, setUser] = useState(null);
  const [activeTab, setActiveTab] = useState('quickYap');
  const [transcripts, setTranscripts] = useState([]);
  const [loadingTranscripts, setLoadingTranscripts] = useState(true);
  const [goals, setGoals] = useState([]);

  // Authentication Setup
  useEffect(() => {
    const initAuth = async () => {
      try {
        if (typeof __initial_auth_token !== 'undefined' && __initial_auth_token) {
          await signInWithCustomToken(auth, __initial_auth_token);
        } else {
          await signInAnonymously(auth);
        }
      } catch (err) {
        console.error("Auth error:", err);
      }
    };
    initAuth();
    const unsubscribe = onAuthStateChanged(auth, setUser);
    return () => unsubscribe();
  }, []);

  // Fetch Transcripts
  useEffect(() => {
    if (!user) return;
    setLoadingTranscripts(true);
    
    const speechesRef = collection(db, 'artifacts', appId, 'users', user.uid, 'speeches');
    // We fetch all and sort in memory per rules to avoid index requirements for complex queries if needed later
    const q = query(speechesRef);

    const unsubscribe = onSnapshot(q, (snapshot) => {
      const data = snapshot.docs.map(doc => ({ id: doc.id, ...doc.data() }));
      // Sort descending by time
      data.sort((a, b) => (b.timestamp?.toMillis() || 0) - (a.timestamp?.toMillis() || 0));
      setTranscripts(data);
      setLoadingTranscripts(false);
    }, (error) => {
      console.error("Firestore error:", error);
      setLoadingTranscripts(false);
    });

    return () => unsubscribe();
  }, [user]);

  // Fetch Goals
  useEffect(() => {
    if (!user) return;
    const goalsRef = collection(db, 'artifacts', appId, 'users', user.uid, 'goals');
    const q = query(goalsRef);

    const unsubscribe = onSnapshot(q, (snapshot) => {
      const data = snapshot.docs.map(doc => ({ id: doc.id, ...doc.data() }));
      data.sort((a, b) => (a.timestamp?.toMillis() || 0) - (b.timestamp?.toMillis() || 0));
      setGoals(data);
    }, (error) => {
      console.error("Firestore error fetching goals:", error);
    });

    return () => unsubscribe();
  }, [user]);

  const saveTranscript = async (data) => {
    if (!user) return;
    try {
      await addDoc(collection(db, 'artifacts', appId, 'users', user.uid, 'speeches'), {
        ...data,
        timestamp: serverTimestamp()
      });
    } catch (err) {
      console.error("Error saving transcript:", err);
    }
  };

  const addGoal = async (text) => {
    if (!user || !text.trim()) return;
    try {
      await addDoc(collection(db, 'artifacts', appId, 'users', user.uid, 'goals'), {
        text: text.trim(),
        timestamp: serverTimestamp()
      });
    } catch (err) {
      console.error("Error adding goal:", err);
    }
  };

  const removeGoal = async (goalId) => {
    if (!user) return;
    try {
      await deleteDoc(doc(db, 'artifacts', appId, 'users', user.uid, 'goals', goalId));
    } catch (err) {
      console.error("Error removing goal:", err);
    }
  };

  if (!user) {
    return <div className="flex h-screen items-center justify-center bg-gray-50 text-gray-800">Authenticating...</div>;
  }

  return (
    <div className="min-h-screen bg-slate-50 text-slate-900 font-sans flex flex-col">
      {/* Header & Navigation */}
      <header className="bg-white border-b border-slate-200 sticky top-0 z-10 shadow-sm">
        <div className="max-w-7xl mx-auto px-4 h-16 flex items-center justify-between">
          <div className="flex items-center gap-2 font-bold text-xl text-indigo-600">
            <Mic className="w-6 h-6" />
            <span>ImprovMaster</span>
          </div>
          <nav className="flex gap-1 sm:gap-4 overflow-x-auto">
            {['quickYap', 'roleplay', 'transcripts'].map((tab) => (
              <button
                key={tab}
                onClick={() => setActiveTab(tab)}
                className={`px-3 py-2 rounded-md text-sm font-medium transition-colors ${
                  activeTab === tab 
                    ? 'bg-indigo-100 text-indigo-700' 
                    : 'text-slate-600 hover:bg-slate-100 hover:text-slate-900'
                }`}
              >
                {tab === 'quickYap' && 'Quick Yap'}
                {tab === 'roleplay' && 'Roleplay'}
                {tab === 'transcripts' && 'Past Transcripts'}
              </button>
            ))}
          </nav>
        </div>
      </header>

      {/* Main Content Area */}
      <div className="flex-1 max-w-7xl mx-auto w-full flex flex-col md:flex-row items-start gap-6 p-4 sm:p-6 md:p-8">
        <aside className="w-full md:w-80 flex-shrink-0">
          <GoalsPanel goals={goals} onAddGoal={addGoal} onRemoveGoal={removeGoal} />
        </aside>
        
        <main className="flex-1 w-full min-w-0">
          {activeTab === 'quickYap' && <QuickYapTab onSave={saveTranscript} goals={goals} />}
          {activeTab === 'roleplay' && <RoleplayTab onSave={saveTranscript} goals={goals} />}
          {activeTab === 'transcripts' && <TranscriptsTab transcripts={transcripts} loading={loadingTranscripts} />}
        </main>
      </div>
    </div>
  );
}

function GoalsPanel({ goals, onAddGoal, onRemoveGoal }) {
  const [newGoal, setNewGoal] = useState('');

  const handleSubmit = (e) => {
    e.preventDefault();
    onAddGoal(newGoal);
    setNewGoal('');
  };

  return (
    <div className="bg-white rounded-xl shadow-sm border border-slate-200 p-5 md:sticky md:top-24">
      <div className="flex items-center gap-2 mb-4 text-indigo-700">
        <Target className="w-5 h-5" />
        <h2 className="text-lg font-bold text-slate-800">My Speech Goals</h2>
      </div>
      
      <form onSubmit={handleSubmit} className="mb-4 flex gap-2">
        <input 
          type="text" 
          value={newGoal}
          onChange={(e) => setNewGoal(e.target.value)}
          placeholder="e.g. Don't use 'um'" 
          className="flex-1 bg-slate-50 border border-slate-200 rounded-lg px-3 py-2 text-sm focus:outline-none focus:ring-2 focus:ring-indigo-500"
        />
        <button 
          type="submit"
          disabled={!newGoal.trim()}
          className="bg-indigo-600 text-white p-2 rounded-lg hover:bg-indigo-700 disabled:opacity-50 transition-colors"
        >
          <Plus className="w-4 h-4" />
        </button>
      </form>

      <div className="space-y-2 max-h-[60vh] overflow-y-auto pr-1">
        {goals.length === 0 ? (
          <p className="text-sm text-slate-500 italic text-center py-4 bg-slate-50 rounded-lg">No goals added yet. Add some to get personalized AI feedback!</p>
        ) : (
          goals.map(goal => (
            <div key={goal.id} className="flex items-center justify-between bg-slate-50 border border-slate-100 rounded-lg p-3 text-sm group transition-colors hover:border-indigo-200">
              <span className="text-slate-700 break-words flex-1 pr-2">{goal.text}</span>
              <button 
                onClick={() => onRemoveGoal(goal.id)}
                className="text-slate-400 hover:text-rose-500 md:opacity-0 group-hover:opacity-100 transition-opacity"
                title="Remove Goal"
              >
                <Trash2 className="w-4 h-4" />
              </button>
            </div>
          ))
        )}
      </div>
    </div>
  );
}

function QuickYapTab({ onSave, goals }) {
  // Phases: idle, spinning, research, organize, speak, grading, finished
  const [phase, setPhase] = useState('idle');
  const [topic, setTopic] = useState('');
  const [timeLeft, setTimeLeft] = useState(0);
  const [transcript, setTranscript] = useState('');
  const [isRecording, setIsRecording] = useState(false);
  const [feedback, setFeedback] = useState(null);
  
  const recognitionRef = useRef(null);

  // Timer Effect
  useEffect(() => {
    let interval = null;
    if (['research', 'organize', 'speak'].includes(phase) && timeLeft > 0) {
      interval = setInterval(() => {
        setTimeLeft(prev => prev - 1);
      }, 1000);
    } else if (timeLeft === 0 && interval) {
      clearInterval(interval);
      advancePhase();
    }
    return () => {
      if (interval) clearInterval(interval);
    };
  }, [phase, timeLeft]);

  const advancePhase = async () => {
    if (phase === 'research') {
      setPhase('organize');
      setTimeLeft(PHASE_TIMERS.organize);
    } else if (phase === 'organize') {
      setPhase('speak');
      setTimeLeft(PHASE_TIMERS.speak);
      setTranscript(''); // Clear previous runs
    } else if (phase === 'speak') {
      if (isRecording) stopRecording();
      setPhase('grading');
      await handleGrading();
    }
  };

  const handleGrading = async () => {
    const result = await evaluateSpeech(topic, transcript, 'quickYap', goals);
    setFeedback(result);
    setPhase('finished');
    onSave({
      type: 'Quick Yap',
      topic: topic,
      transcript: transcript || '(No transcript recorded)',
      grade: result.grade,
      summary: result.summary
    });
  };

  const spinTopic = () => {
    setPhase('spinning');
    let spins = 0;
    const maxSpins = 20;
    const interval = setInterval(() => {
      setTopic(TOPICS[Math.floor(Math.random() * TOPICS.length)]);
      spins++;
      if (spins >= maxSpins) {
        clearInterval(interval);
        setTimeout(() => {
          setPhase('research');
          setTimeLeft(PHASE_TIMERS.research);
        }, 500);
      }
    }, 50);
  };

  // Web Speech API Setup
  const toggleRecording = () => {
    if (isRecording) {
      stopRecording();
    } else {
      startRecording();
    }
  };

  const startRecording = () => {
    const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
    if (!SpeechRecognition) {
      // alert replacement
      console.error("Speech Recognition not supported in this browser.");
      return;
    }

    if (!recognitionRef.current) {
      recognitionRef.current = new SpeechRecognition();
      recognitionRef.current.continuous = true;
      recognitionRef.current.interimResults = true;
      
      recognitionRef.current.onresult = (event) => {
        let finalTranscript = '';
        let interimTranscript = '';
        for (let i = event.resultIndex; i < event.results.length; ++i) {
          if (event.results[i].isFinal) {
            finalTranscript += event.results[i][0].transcript + ' ';
          } else {
            interimTranscript += event.results[i][0].transcript;
          }
        }
        setTranscript(prev => prev + finalTranscript);
      };

      recognitionRef.current.onend = () => {
        // Auto-restart if we are still supposed to be recording
        if (isRecording && phase === 'speak') {
          try {
             recognitionRef.current.start();
          } catch(e) {}
        } else {
          setIsRecording(false);
        }
      };
    }

    try {
      recognitionRef.current.start();
      setIsRecording(true);
    } catch (e) {
      console.error(e);
    }
  };

  const stopRecording = () => {
    if (recognitionRef.current) {
      recognitionRef.current.stop();
      setIsRecording(false);
    }
  };

  const reset = () => {
    setPhase('idle');
    setTopic('');
    setTranscript('');
    setFeedback(null);
    if (isRecording) stopRecording();
  };

  return (
    <div className="bg-white rounded-xl shadow-sm border border-slate-200 p-6 sm:p-10 max-w-2xl mx-auto">
      <div className="text-center mb-8">
        <h1 className="text-3xl font-bold text-slate-800 mb-2">Quick Yap</h1>
        <p className="text-slate-500">Practice thinking on your feet with random topics.</p>
      </div>

      {phase === 'idle' && (
        <div className="flex flex-col items-center justify-center py-12">
          <div className="w-24 h-24 bg-indigo-50 rounded-full flex items-center justify-center mb-6">
            <RefreshCw className="w-10 h-10 text-indigo-500" />
          </div>
          <button 
            onClick={spinTopic}
            className="bg-indigo-600 hover:bg-indigo-700 text-white font-semibold py-3 px-8 rounded-full shadow-md transition-transform active:scale-95"
          >
            Spin a Topic
          </button>
        </div>
      )}

      {phase === 'spinning' && (
        <div className="flex flex-col items-center justify-center py-12">
          <div className="text-2xl font-medium text-slate-700 animate-pulse text-center">
            {topic}
          </div>
        </div>
      )}

      {['research', 'organize', 'speak'].includes(phase) && (
        <div className="flex flex-col items-center">
          <div className="bg-slate-50 w-full p-6 rounded-lg border border-slate-100 mb-8 text-center">
            <h2 className="text-sm font-semibold text-indigo-600 uppercase tracking-wider mb-2">Your Topic</h2>
            <p className="text-2xl font-medium text-slate-800">{topic}</p>
          </div>

          <div className="flex flex-col items-center mb-8">
            <div className="text-sm font-semibold text-slate-500 uppercase tracking-wider mb-2">
              {phase === 'research' && "Research Phase"}
              {phase === 'organize' && "Organize Thoughts Phase"}
              {phase === 'speak' && "Speaking Phase"}
            </div>
            <div className={`text-6xl font-bold font-mono tracking-tighter ${timeLeft <= 10 ? 'text-rose-500 animate-pulse' : 'text-slate-800'}`}>
              {formatTime(timeLeft)}
            </div>
            <button 
              onClick={advancePhase}
              className="mt-4 flex items-center gap-2 text-sm text-slate-500 hover:text-indigo-600 transition-colors"
            >
              <SkipForward className="w-4 h-4" /> Skip Timer
            </button>
          </div>

          {phase === 'speak' && (
            <div className="w-full flex flex-col items-center gap-4">
              <button
                onClick={toggleRecording}
                className={`flex items-center gap-2 py-3 px-6 rounded-full font-semibold shadow-md transition-all ${
                  isRecording 
                    ? 'bg-rose-100 text-rose-700 hover:bg-rose-200 border border-rose-200 animate-pulse' 
                    : 'bg-emerald-600 text-white hover:bg-emerald-700'
                }`}
              >
                {isRecording ? <Square className="w-5 h-5 fill-current" /> : <Mic className="w-5 h-5" />}
                {isRecording ? 'Stop Recording' : 'Start Recording'}
              </button>
              
              <div className="w-full p-4 bg-slate-50 border border-slate-200 rounded-lg min-h-[120px] max-h-[200px] overflow-y-auto">
                {transcript || <span className="text-slate-400 italic">Your speech transcript will appear here...</span>}
              </div>
            </div>
          )}
        </div>
      )}

      {phase === 'grading' && (
        <div className="flex flex-col items-center justify-center py-12">
          <RefreshCw className="w-10 h-10 text-indigo-500 animate-spin mb-4" />
          <p className="text-lg font-medium text-slate-700">AI Coach is evaluating your speech...</p>
        </div>
      )}

      {phase === 'finished' && feedback && (
        <div className="animate-in fade-in slide-in-from-bottom-4 duration-500">
          <div className="bg-emerald-50 border border-emerald-100 rounded-xl p-6 text-center mb-6">
            <div className="w-16 h-16 bg-emerald-100 text-emerald-600 rounded-full flex items-center justify-center mx-auto mb-4 text-3xl font-bold">
              {feedback.grade}
            </div>
            <h3 className="text-xl font-bold text-slate-800 mb-2">Evaluation Complete</h3>
            <p className="text-slate-600">{feedback.summary}</p>
          </div>
          
          <div className="mb-6">
            <h4 className="text-sm font-semibold text-slate-500 uppercase mb-2">Transcript</h4>
            <div className="p-4 bg-slate-50 border border-slate-200 rounded-lg text-slate-700 max-h-40 overflow-y-auto">
              {transcript || '(No transcript recorded)'}
            </div>
          </div>

          <div className="flex justify-center">
            <button 
              onClick={reset}
              className="bg-indigo-600 hover:bg-indigo-700 text-white font-semibold py-3 px-8 rounded-full shadow-md transition-colors"
            >
              Go Again
            </button>
          </div>
        </div>
      )}
    </div>
  );
}

function RoleplayTab({ onSave, goals }) {

  const [timeLeft, setTimeLeft] = useState(0);
  const [transcript, setTranscript] = useState('');
  const [isRecording, setIsRecording] = useState(false);
  const [feedback, setFeedback] = useState(null);
  
  const recognitionRef = useRef(null);

  useEffect(() => {
    let interval = null;
    if (phase === 'speak' && timeLeft > 0) {
      interval = setInterval(() => {
        setTimeLeft(prev => prev - 1);
      }, 1000);
    } else if (timeLeft === 0 && interval) {
      clearInterval(interval);
      handleGrading();
    }
    return () => {
      if (interval) clearInterval(interval);
    };
  }, [phase, timeLeft]);

    setPhase('spinning');

    const maxSpins = 15;

      setScenario(ROLEPLAY_SCENARIOS[Math.floor(Math.random() * ROLEPLAY_SCENARIOS.length)]);

          setPhase('speak');
          setTimeLeft(PHASE_TIMERS.roleplay);
          setTranscript('');
        }, 500);
      }
    }, 60);
  };

  const handleGrading = async () => {
    if (isRecording) stopRecording();
    setPhase('grading');
    const result = await evaluateSpeech(scenario, transcript, 'roleplay', goals);
    setFeedback(result);
    setPhase('finished');
    onSave({
      type: 'Roleplay',
      topic: scenario,
      transcript: transcript || '(No transcript recorded)',
      grade: result.grade,
      summary: result.summary
    });
  };

  // Web Speech API Setup
  const toggleRecording = () => {
    if (isRecording) {
      stopRecording();
    } else {
      startRecording();
    }
  };

  const startRecording = () => {
    const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
    if (!SpeechRecognition) return;

    if (!recognitionRef.current) {
      recognitionRef.current = new SpeechRecognition();
      recognitionRef.current.continuous = true;
      recognitionRef.current.interimResults = true;
      
      recognitionRef.current.onresult = (event) => {
        let finalTranscript = '';
        for (let i = event.resultIndex; i < event.results.length; ++i) {
          if (event.results[i].isFinal) {
            finalTranscript += event.results[i][0].transcript + ' ';
          }
        }
        setTranscript(prev => prev + finalTranscript);
      };

      recognitionRef.current.onend = () => {
        if (isRecording && phase === 'speak') {
          try { recognitionRef.current.start(); } catch(e) {}
        } else {
          setIsRecording(false);
        }
      };
    }

    try {
      recognitionRef.current.start();
      setIsRecording(true);
    } catch (e) {}
  };

  const stopRecording = () => {
    if (recognitionRef.current) {
      recognitionRef.current.stop();
      setIsRecording(false);
    }
  };

  const reset = () => {
    setPhase('idle');
    setScenario('');
    setTranscript('');
    setFeedback(null);
    if (isRecording) stopRecording();
  };

  return (
    <div className="bg-white rounded-xl shadow-sm border border-slate-200 p-6 sm:p-10 max-w-2xl mx-auto">
      <div className="text-center mb-8">
        <h1 className="text-3xl font-bold text-slate-800 mb-2">Roleplay Scenario</h1>
        <p className="text-slate-500">Solve problems and practice interpersonal skills under pressure.</p>
      </div>

      {phase === 'idle' && (
        <div className="flex flex-col items-center justify-center py-12">
          <div className="w-24 h-24 bg-orange-50 rounded-full flex items-center justify-center mb-6">
            <MessageSquare className="w-10 h-10 text-orange-500" />
          </div>
          <button 
            onClick={spinScenario}
            className="bg-orange-500 hover:bg-orange-600 text-white font-semibold py-3 px-8 rounded-full shadow-md transition-transform active:scale-95"
          >
            Get Scenario
          </button>
        </div>
      )}

      {phase === 'spinning' && (
        <div className="flex flex-col items-center justify-center py-12">
          <div className="text-xl font-medium text-slate-700 animate-pulse text-center p-6 bg-slate-50 rounded-lg">
            {scenario}
          </div>
        </div>
      )}

      {phase === 'speak' && (
        <div className="flex flex-col items-center">
          <div className="bg-orange-50 w-full p-6 rounded-lg border border-orange-100 mb-8">
            <h2 className="text-sm font-semibold text-orange-600 uppercase tracking-wider mb-3 flex items-center gap-2">
              <CheckCircle className="w-4 h-4" /> Scenario
            </h2>
            <p className="text-lg font-medium text-slate-800 italic">"{scenario}"</p>
          </div>

          <div className="flex flex-col items-center mb-8">
            <div className="text-sm font-semibold text-slate-500 uppercase tracking-wider mb-2">Time to Respond</div>
            <div className={`text-6xl font-bold font-mono tracking-tighter ${timeLeft <= 10 ? 'text-rose-500 animate-pulse' : 'text-slate-800'}`}>
              {formatTime(timeLeft)}
            </div>
            <button 
              onClick={handleGrading}
              className="mt-4 flex items-center gap-2 text-sm text-slate-500 hover:text-orange-600 transition-colors"
            >
              <SkipForward className="w-4 h-4" /> Finish Early
            </button>
          </div>

          <div className="w-full flex flex-col items-center gap-4">
            <button
              onClick={toggleRecording}
              className={`flex items-center gap-2 py-3 px-6 rounded-full font-semibold shadow-md transition-all ${
                isRecording 
                  ? 'bg-rose-100 text-rose-700 hover:bg-rose-200 border border-rose-200 animate-pulse' 
                  : 'bg-emerald-600 text-white hover:bg-emerald-700'
              }`}
            >
              {isRecording ? <Square className="w-5 h-5 fill-current" /> : <Mic className="w-5 h-5" />}
              {isRecording ? 'Stop Recording' : 'Start Recording'}
            </button>
            
            <div className="w-full p-4 bg-slate-50 border border-slate-200 rounded-lg min-h-[120px] max-h-[200px] overflow-y-auto">
              {transcript || <span className="text-slate-400 italic">Speak your solution...</span>}
            </div>
          </div>
        </div>
      )}

      {phase === 'grading' && (
        <div className="flex flex-col items-center justify-center py-12">
          <RefreshCw className="w-10 h-10 text-orange-500 animate-spin mb-4" />
          <p className="text-lg font-medium text-slate-700">AI Coach is evaluating your response...</p>
        </div>
      )}

      {phase === 'finished' && feedback && (
        <div className="animate-in fade-in slide-in-from-bottom-4 duration-500">
          <div className="bg-emerald-50 border border-emerald-100 rounded-xl p-6 text-center mb-6">
            <div className="w-16 h-16 bg-emerald-100 text-emerald-600 rounded-full flex items-center justify-center mx-auto mb-4 text-3xl font-bold">
              {feedback.grade}
            </div>
            <h3 className="text-xl font-bold text-slate-800 mb-2">Evaluation Complete</h3>
            <p className="text-slate-600">{feedback.summary}</p>
          </div>
          
          <div className="mb-6">
            <h4 className="text-sm font-semibold text-slate-500 uppercase mb-2">Your Answer</h4>
            <div className="p-4 bg-slate-50 border border-slate-200 rounded-lg text-slate-700 max-h-40 overflow-y-auto">
              {transcript || '(No transcript recorded)'}
            </div>
          </div>

          <div className="flex justify-center">
            <button 
              onClick={reset}
              className="bg-orange-500 hover:bg-orange-600 text-white font-semibold py-3 px-8 rounded-full shadow-md transition-colors"
            >
              Try Another Scenario
            </button>
          </div>
        </div>
      )}
    </div>
  );
}

function TranscriptsTab({ transcripts, loading }) {
  if (loading) {
    return (
      <div className="flex justify-center items-center py-20">
        <RefreshCw className="w-8 h-8 text-indigo-500 animate-spin" />
      </div>
    );
  }

  if (transcripts.length === 0) {
    return (
      <div className="bg-white rounded-xl shadow-sm border border-slate-200 p-10 text-center max-w-2xl mx-auto">
        <FileText className="w-16 h-16 text-slate-300 mx-auto mb-4" />
        <h2 className="text-2xl font-bold text-slate-800 mb-2">No past transcripts</h2>
        <p className="text-slate-500">Do a Quick Yap or Roleplay to start building your record!</p>
      </div>
    );
  }

  return (
    <div className="grid grid-cols-1 md:grid-cols-2 gap-6">
      {transcripts.map((item) => (
        <div key={item.id} className="bg-white rounded-xl shadow-sm border border-slate-200 flex flex-col overflow-hidden transition-all hover:shadow-md">
          <div className="bg-slate-50 px-5 py-4 border-b border-slate-100 flex justify-between items-center">
            <span className={`text-xs font-bold uppercase tracking-wider px-2 py-1 rounded-full ${
              item.type === 'Quick Yap' ? 'bg-indigo-100 text-indigo-700' : 'bg-orange-100 text-orange-700'
            }`}>
              {item.type}
            </span>
            <div className="flex items-center gap-1 font-bold text-emerald-600 bg-emerald-50 px-2 py-1 rounded-md">
              <Award className="w-4 h-4" /> Grade: {item.grade}
            </div>
          </div>
          
          <div className="p-5 flex-1 flex flex-col">
            <h3 className="font-semibold text-slate-800 mb-3 line-clamp-2" title={item.topic}>
              {item.topic}
            </h3>
            
            <div className="text-sm text-slate-600 bg-slate-50 p-3 rounded-lg mb-4 italic flex-1 max-h-32 overflow-y-auto">
              "{item.transcript}"
            </div>
            
            <div>
              <h4 className="text-xs font-semibold text-slate-400 uppercase mb-1">AI Feedback</h4>
              <p className="text-sm text-slate-700">{item.summary}</p>
            </div>
          </div>
          
          {item.timestamp && (

              {new Date(item.timestamp.toMillis()).toLocaleString()}
            </div>
          )}
        </div>
      ))}
    </div>
  );
}
